# --- SET YOUR TUNING HERE ---
speed=1.33
zoom=1.15

# --- AUTOMATED CALCULATIONS ---
# Video speed multiplier (1/speed)
setpts=$(echo "scale=4; 1/$speed" | bc)
# Zoomed width (1080 * zoom)
z_width=$(echo "1080 * $zoom" | bc | cut -d'.' -f1)

# --- THE COMMAND ---
ffmpeg -i "D:/youtube_slicer_output/SHORT_RESIDENT_EVIL_8_VILLAGE_final_result_2026-02-02_18-34-20_peleando contra la hija cachonda de Lady Dimitrescu.webm" \
  -vf "setpts=${setpts}*PTS,split[original][copy];[copy]scale=270:480:force_original_aspect_ratio=increase,crop=270:480,boxblur=7:1,scale=1080:1920[blurred];[original]scale=${z_width}:-1,crop=1080:ih[sharp];[blurred][sharp]overlay=(W-w)/2:(H-h)/2" \
  -af "atempo=${speed}" \
  -c:v libx264 -crf 18 -preset veryfast \
